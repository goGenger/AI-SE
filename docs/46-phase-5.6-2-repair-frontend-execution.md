/* Generated by the ai-se frontend agent for visual:fixture-page:1. */
import { Button } from "../components/Button.js";

export function renderFixturePage() {
  return `
  <section class="visual-section" data-section-id="section:header:1"
    data-rect="0,0,1440,180"
    style="min-height: 180px; color: var(--colors-primary); font-size: var(--typography-body); border-radius: var(--radii-card);">
    <h2>Header</h2>
    <span data-element-id="element:logo:1" data-missing-component="Icon">logo: icon</span>

  `tokens.css` 一致(colors.primary=#1533ff 由 5.6-1 真实分析产生);
